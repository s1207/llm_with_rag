# 1️⃣ Defect Region Extraction (from PatchCore heatmaps)

**File:** `defect_extraction.py`

```python
import cv2
import numpy as np


def extract_defect_patches(image, anomaly_map,
                            thresh=0.5,
                            min_area=200,
                            pad=8):
    """
    image: grayscale or RGB image
    anomaly_map: PatchCore heatmap (normalized 0–1)
    """

    H, W = anomaly_map.shape
    mask = (anomaly_map > thresh).astype(np.uint8) * 255

    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)

    nlabels, labels, stats, _ = cv2.connectedComponentsWithStats(mask)

    patches, boxes = [], []

    for i in range(1, nlabels):
        x, y, w, h, area = stats[i]

        if area < min_area:
            continue

        x0 = max(0, x - pad)
        y0 = max(0, y - pad)
        x1 = min(W, x + w + pad)
        y1 = min(H, y + h + pad)

        crop = image[y0:y1, x0:x1]

        if crop.size == 0:
            continue

        patches.append(crop)
        boxes.append((x0, y0, x1 - x0, y1 - y0))

    return patches, boxes
```

# 2️⃣ Context Feature Computation

**File:** `context_features.py`

```python
import torch
import numpy as np


def compute_context_features(box, anomaly_map):
    x, y, w, h = box

    patch_scores = anomaly_map[y:y+h, x:x+w]

    area = (w * h) / (anomaly_map.shape[0] * anomaly_map.shape[1])
    aspect_ratio = w / max(h, 1)
    mean_score = patch_scores.mean()
    max_score = patch_scores.max()

    return torch.tensor([
        area,
        aspect_ratio,
        mean_score,
        max_score
    ], dtype=torch.float32)
```

# 3️⃣ Mahalanobis + Prototype Classifier (Drop-in Replacement)

**File:** `classifier.py` (Replace your existing classifier with this)

```python
import torch
import numpy as np


class DefectClassifier:
    def __init__(self, reg=1e-4):
        self.reg = reg
        self.class_means = {}
        self.class_covs = {}
        self.class_inv_covs = {}

    def fit(self, features, labels):
        """
        features: NxD tensor
        labels: N list or tensor
        """
        classes = sorted(set(labels))

        for c in classes:
            feats = features[torch.tensor(labels) == c]
            mean = feats.mean(0)

            cov = torch.cov(feats.T)
            cov += self.reg * torch.eye(cov.shape[0])

            inv_cov = torch.inverse(cov)

            self.class_means[c] = mean
            self.class_covs[c] = cov
            self.class_inv_covs[c] = inv_cov

    def predict(self, x):
        """
        x: Dx1 tensor
        """
        scores = {}

        for c in self.class_means:
            diff = x - self.class_means[c]
            dist = torch.sqrt(diff @ self.class_inv_covs[c] @ diff)
            scores[c] = dist.item()

        return min(scores, key=scores.get), scores
```

# 4️⃣ End-to-End Inference Pipeline

**File:** `inference_pipeline.py`

```python
import cv2
import torch
from defect_extraction import extract_defect_patches
from context_features import compute_context_features


@torch.no_grad()
def run_full_inference(
        image,
        anomaly_map,
        feature_extractor,
        classifier,
        device="cuda",
        thresh=0.5
    ):

    patches, boxes = extract_defect_patches(
        image, anomaly_map, thresh=thresh)

    results = []

    for patch, box in zip(patches, boxes):
        patch = cv2.resize(patch, (224, 224))
        patch = torch.tensor(patch).float().unsqueeze(0).unsqueeze(0) / 255.0
        patch = patch.to(device)

        emb = feature_extractor(patch).squeeze()

        ctx = compute_context_features(box, anomaly_map).to(device)

        feat = torch.cat([emb, ctx], dim=0)

        cls, scores = classifier.predict(feat)

        results.append({
            "class": cls,
            "bbox": box,
            "scores": scores
        })

    return results
```

**File:** `visualization.py`

```python
import cv2


def draw_predictions(image, predictions):
    img = cv2.cvtColor(image, cv2.COLOR_GRAY2BGR)

    for pred in predictions:
        x,y,w,h = pred["bbox"]
        label = pred["class"]

        cv2.rectangle(img, (x,y), (x+w,y+h), (0,255,0), 2)
        cv2.putText(img, label, (x,y-10),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0,255,0), 2)

    return img
```
